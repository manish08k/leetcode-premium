
select 
    s1.user_id, 
    s1.steps_date, 
    round((s1.steps_count + s2.steps_count + s3.steps_count) / 3, 2) as rolling_average 
from steps as s1
join steps as s2
on s1.user_id = s2.user_id
and datediff(s1.steps_date, s2.steps_date) = 1
join steps as s3
on s1.user_id = s3.user_id
and datediff(s1.steps_date, s3.steps_date) = 2
order by s1.user_id, s1.steps_date

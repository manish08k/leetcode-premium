with cte as (
select product_id ,year(purchase_date)  purchase_date
from orders 
group by product_id ,year(purchase_date)
having count(*)>=3 )



select distinct  c1.product_id
from cte c1 ,cte c2 
where c1.product_id =c2.product_id and abs(c1.purchase_date -c2.purchase_date)=1

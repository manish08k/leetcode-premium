with cte as (
    select
        distinct
        id,
        login_date
    from logins
), cte2 as (
    select
        id,
        login_date,
        DATE_SUB(login_date, INTERVAL row_number() over(partition by id order by login_date) DAY) as grp
    from cte
    order by id, login_date
), cte3 as (
    select
        distinct id
    from cte2
    group by id, grp
    having count(*) >=5
)
select
    cte3.id,
    name
from cte3 join accounts as a on cte3.id=a.id
order by id
;

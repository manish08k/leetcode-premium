WITH cte AS (
    SELECT
        book_id
    FROM Books
    WHERE book_id NOT IN (
        SELECT
            book_id
        FROM Books
        WHERE available_from BETWEEN '2019-05-23' AND '2019-06-23'
    )
)
SELECT
    c.book_id,
    b.name
FROM cte c
JOIN Books b
    ON b.book_id = c.book_id
WHERE c.book_id NOT IN (
    SELECT
        book_id
    FROM Orders
    WHERE dispatch_date BETWEEN '2018-06-23' AND '2019-06-23'
    GROUP BY book_id
    HAVING SUM(quantity) >= 10
)
